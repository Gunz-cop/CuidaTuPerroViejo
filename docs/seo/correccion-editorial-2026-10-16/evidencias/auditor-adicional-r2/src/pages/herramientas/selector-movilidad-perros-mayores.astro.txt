// @ts-nocheck -- script de UI legado, sin tipos; la lógica tipada vive en src/lib/agent-tools/.
import { HANDLING, LEVELS, RESULTS, resultKey } from '../../lib/agent-tools/movilidad.ts';
import { AFFILIATE_URLS } from '../../lib/agent-tools/movilidad-afiliados.ts';

(function () {
  'use strict';

  /* ── Enlaces de afiliados: ocultos por el momento ── */
  var SHOW_AFFILIATE_LINKS = false;

  /* ── Tema visual por nivel de severidad ── */
  var THEME = {
    L: { icon: 'bg-green-500/10', badge: 'bg-green-500/15 text-green-700 dark:text-green-300' },
    M: { icon: 'bg-amber-500/10', badge: 'bg-amber-500/15 text-amber-700 dark:text-amber-300' },
    S: { icon: 'bg-red-500/10',   badge: 'bg-red-500/15 text-red-700 dark:text-red-300' }
  };

  /* ── Textos ── */
  var LB1 = 'Nivel de movilidad';
  var LB2 = 'Car\u00e1cter con su cuerpo';
  var QT1 = '\u00bfQu\u00e9 ayuda necesita tu perro hoy?';
  var QT2 = '\u00bfC\u00f3mo es con sus patas y cuerpo?';

  /* ── Opciones y resultados: src/lib/agent-tools/movilidad.ts (compartidos con la API y WebMCP) ── */
  var OPT1 = LEVELS.map(function (o) {
    return { ico: o.icon, tx: o.title, sub: o.subtitle, v: o.code, lvl: o.code };
  });

  var OPT2 = HANDLING.map(function (o) {
    return { ico: o.icon, tx: o.title, sub: o.subtitle, v: o.code, lvl: '' };
  });

  /* ── Datos de resultados (mismas claves LA..SB que la UI legada) ── */
  var RES = {};
  Object.keys(RESULTS).forEach(function (key) {
    var r = RESULTS[key];
    RES[key] = {
      ico: r.icon,
      badge: r.badge,
      tit: r.title,
      desc: r.description,
      prods: r.plan.map(function (p, i) { return { name: p.name, why: p.why, url: AFFILIATE_URLS[key][i] }; })
    };
  });

  /* ── Estado ── */
  var a1 = null, a2 = null;

  /* ── DOM (cacheado en init) ── */
  var D = {};

  function init() {
    D.s0     = document.getElementById('m5s0');
    D.s1     = document.getElementById('m5s1');
    D.s2     = document.getElementById('m5s2');
    D.s3     = document.getElementById('m5s3');
    D.start  = document.getElementById('m5start');
    D.b1     = document.getElementById('m5b1');
    D.b2     = document.getElementById('m5b2');
    D.pb1    = document.getElementById('m5pb1');
    D.pb2    = document.getElementById('m5pb2');
    D.lb1    = document.getElementById('m5lb1');
    D.lb2    = document.getElementById('m5lb2');
    D.qt1    = document.getElementById('m5qt1');
    D.qt2    = document.getElementById('m5qt2');
    D.oo1    = document.getElementById('m5oo1');
    D.oo2    = document.getElementById('m5oo2');
    D.alert  = document.getElementById('m5alert');
    D.ricowr = document.getElementById('m5ricowrap');
    D.rico   = document.getElementById('m5rico');
    D.rbadge = document.getElementById('m5rbadge');
    D.rtit   = document.getElementById('m5rtit');
    D.rdesc  = document.getElementById('m5rdesc');
    D.prods  = document.getElementById('m5prods');
    D.rst    = document.getElementById('m5rst');
    D.copy   = document.getElementById('m5copy');
    D.wa     = document.getElementById('m5wa');
    D.copyok = document.getElementById('m5copyok');
    D.stickybar  = document.getElementById('m5stickybar');
    D.rstFixed   = document.getElementById('m5rstFixed');
    D.waFixed    = document.getElementById('m5waFixed');

    /* Textos de preguntas */
    D.lb1.textContent = LB1;
    D.lb2.textContent = LB2;
    D.qt1.textContent = QT1;
    D.qt2.textContent = QT2;

    /* Barras de progreso */
    D.pb1.style.width = '50%';
    D.pb2.style.width = '100%';

    /* Construir opciones */
    mkOpts(D.oo1, OPT1, function (v) { a1 = v; go(D.s1, D.s2); });
    mkOpts(D.oo2, OPT2, function (v) { a2 = v; showResult(resultKey(a1, a2)); });

    /* Navegación */
    D.start.addEventListener('click', function () { go(D.s0, D.s1); });
    D.b1.addEventListener('click',    function () { go(D.s1, D.s0); });
    D.b2.addEventListener('click',    function () { go(D.s2, D.s1); a1 = null; });
    D.rst.addEventListener('click',   reset);
    D.copy.addEventListener('click',  copyResult);
    D.wa.addEventListener('click',    shareWhatsApp);
    D.rstFixed.addEventListener('click', reset);
    D.waFixed.addEventListener('click',  shareWhatsApp);
  }

  /* ── Mantener el widget a la vista en cada paso (clave en móvil) ── */
  function scrollToWidget() {
    var el = document.getElementById('m5widget');
    if (!el) return;
    el.scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  /* ── Transición ── */
  function go(from, to) {
    from.classList.add('off');
    to.classList.remove('off');
    to.classList.remove('m5anim');
    void to.offsetWidth;
    to.classList.add('m5anim');
    scrollToWidget();
  }

  /* ── Construir botones de opción ── */
  function mkOpts(container, opts, cb) {
    opts.forEach(function (o) {
      var btn = document.createElement('button');
      btn.type = 'button';
      
      // Dynamic button with beautiful Tailwind classes
      btn.className = 'w-full min-h-[72px] p-4 md:p-5 bg-brand-card/50 dark:bg-slate-900/35 hover:bg-brand-card-hover active:bg-brand-card-hover dark:hover:bg-slate-900/60 dark:active:bg-slate-900/60 border-2 border-transparent focus:border-brand-primary hover:border-brand-primary/60 active:border-brand-primary rounded-2xl flex items-center gap-4 text-left transition-all duration-150 outline-none select-none cursor-pointer hover:shadow-md hover:-translate-y-0.5 active:scale-[0.98]';
      
      if (o.lvl === 'S') {
        btn.addEventListener('mouseenter', function() { btn.style.borderColor = '#b71c1c'; });
        btn.addEventListener('mouseleave', function() { btn.style.borderColor = 'transparent'; });
        btn.addEventListener('focus', function() { btn.style.borderColor = '#b71c1c'; });
        btn.addEventListener('blur', function() { btn.style.borderColor = 'transparent'; });
      }
      
      btn.setAttribute('aria-label', o.tx + (o.sub ? ': ' + o.sub : ''));

      var ico = document.createElement('span');
      ico.className = 'flex items-center justify-center w-11 h-11 shrink-0 rounded-xl bg-brand-secondary/50 dark:bg-slate-800/60 text-xl select-none';
      ico.setAttribute('aria-hidden', 'true');
      ico.textContent = o.ico;

      var body = document.createElement('span');
      body.className = 'flex-1 flex flex-col';

      var tx = document.createElement('span');
      tx.className = 'font-semibold text-brand-text text-sm leading-snug';
      tx.textContent = o.tx;
      body.appendChild(tx);

      if (o.sub) {
        var sub = document.createElement('span');
        sub.className = 'text-xs text-brand-muted mt-1 leading-normal font-normal';
        sub.textContent = o.sub;
        body.appendChild(sub);
      }

      var arr = document.createElement('span');
      arr.className = 'text-xl text-brand-muted/40 font-light select-none ml-auto';
      arr.setAttribute('aria-hidden', 'true');
      arr.textContent = '\u203a'; // '›'

      btn.appendChild(ico);
      btn.appendChild(body);
      btn.appendChild(arr);
      btn.addEventListener('click', function () { cb(o.v); });
      container.appendChild(btn);
    });
  }

  /* ── Mostrar resultado ── */
  var lastResult = null;

  function showResult(key) {
    var r = RES[key];
    if (!r) return;
    lastResult = r;

    /* Alerta paliativa: visible solo si nivel S */
    if (a1 === 'S') {
      D.alert.classList.remove('off');
    } else {
      D.alert.classList.add('off');
    }

    /* Tema visual por severidad */
    var theme = THEME[a1] || THEME.L;
    D.ricowr.className = 'inline-flex items-center justify-center w-16 h-16 rounded-2xl text-3xl mb-3 ' + theme.icon;
    D.rbadge.className = 'inline-block text-xs font-bold tracking-wider uppercase px-3 py-1 rounded-full mb-3 ' + theme.badge;

    D.rico.textContent   = r.ico;
    D.rbadge.textContent = r.badge;
    D.rtit.textContent   = r.tit;
    D.rdesc.textContent  = r.desc;

    /* Construir el plan de acción (sin enlaces de compra por el momento) */
    D.prods.innerHTML = '';
    r.prods.forEach(function (p, i) {
      var card = document.createElement('div');
      card.className = 'flex items-start gap-4 bg-brand-card/30 dark:bg-slate-900/15 border border-brand-border/40 rounded-2xl p-4 md:p-5 shadow-sm';

      var num = document.createElement('div');
      num.className = 'shrink-0 flex items-center justify-center w-8 h-8 rounded-full bg-brand-primary/10 text-brand-primary font-bold text-sm select-none';
      num.setAttribute('aria-hidden', 'true');
      num.textContent = String(i + 1);

      var body = document.createElement('div');
      body.className = 'flex-1 flex flex-col gap-1';

      var name = document.createElement('div');
      name.className = 'font-bold text-brand-text text-base';
      name.textContent = p.name;

      var why = document.createElement('div');
      why.className = 'text-sm text-brand-muted leading-relaxed';
      why.textContent = p.why;

      body.appendChild(name);
      body.appendChild(why);
      if (SHOW_AFFILIATE_LINKS && p.url) {
      var link = document.createElement('a');
      link.className = 'inline-flex self-start items-center gap-1.5 rounded-full border border-brand-primary px-4 py-1.5 text-xs font-bold text-brand-primary hover:bg-brand-primary hover:text-white transition-all no-underline cursor-pointer';
      link.href = p.url;
      link.target = '_blank';
      link.rel = 'noopener sponsored';
      link.textContent = 'Ver en AliExpress';
      body.appendChild(link);
      }

      card.appendChild(num);
      card.appendChild(body);
      D.prods.appendChild(card);
    });

    go(D.s2, D.s3);
    D.stickybar.classList.remove('off');
    document.body.classList.add('m5-result-active');
  }

  /* ── Texto plano del resultado (para copiar / compartir) ── */
  function resultText() {
    if (!lastResult) return '';
    var lines = ['Resultado: ' + lastResult.tit, '', lastResult.desc, '', 'Plan de acción:'];
    lastResult.prods.forEach(function (p, i) {
      lines.push('  ' + (i + 1) + '. ' + p.name + ' - ' + p.why);
    });
    lines.push('');
    lines.push('https://cuidatuperroviejo.com/herramientas/selector-movilidad-perros-mayores');
    return lines.join('\n');
  }

  /* ── Copiar resultado ── */
  function copyResult() {
    var text = resultText();
    if (!text || !navigator.clipboard) return;
    navigator.clipboard.writeText(text).then(function () {
      if (D.copyok) {
        D.copyok.style.display = 'block';
        setTimeout(function () { D.copyok.style.display = 'none'; }, 3000);
      }
    }).catch(function (err) {
      console.error('Error al copiar: ', err);
    });
  }

  /* ── Compartir por WhatsApp ── */
  function shareWhatsApp() {
    var text = resultText();
    if (!text) return;
    window.open('https://wa.me/?text=' + encodeURIComponent(text), '_blank');
  }

  /* ── Reset ── */
  function reset() {
    a1 = null;
    a2 = null;
    D.s1.classList.add('off');  D.s1.classList.remove('m5anim');
    D.s2.classList.add('off');  D.s2.classList.remove('m5anim');
    D.s3.classList.add('off');  D.s3.classList.remove('m5anim');
    D.s0.classList.remove('off');
    D.s0.classList.remove('m5anim');
    void D.s0.offsetWidth;
    D.s0.classList.add('m5anim');
    D.stickybar.classList.add('off');
    document.body.classList.remove('m5-result-active');
    scrollToWidget();
  }

  /* ── Arranque robusto para Blogger/Astro ── */
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }

}());
</script>
