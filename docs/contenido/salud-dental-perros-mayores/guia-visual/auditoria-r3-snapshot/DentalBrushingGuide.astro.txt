<script>
  class DentalGuide {
    private step = 0;
    private playing = false;
    private timer: ReturnType<typeof setTimeout> | undefined;
    private abort = new AbortController();
    private reduced = matchMedia('(prefers-reduced-motion: reduce)');
    constructor(private root: HTMLElement) {
      root.dataset.enhanced = 'true';
      root.querySelectorAll<HTMLElement>('.guide-tabs,.guide-controls,[data-detail-toggle]').forEach(el => el.hidden = false);
      root.querySelectorAll<HTMLButtonElement>('[data-step]').forEach(button => button.addEventListener('click', () => { this.stop(); this.step = Number(button.dataset.step); this.render(); }, { signal: this.abort.signal }));
      root.querySelectorAll<HTMLButtonElement>('[data-action]').forEach(button => button.addEventListener('click', () => {
        if (button.dataset.action === 'play') {
          if (this.playing) this.stop();
          else { if (this.step === 3) this.step = 0; this.playing = true; this.render(); this.schedule(); }
        } else { this.stop(); this.step = Math.max(0, Math.min(3, this.step + (button.dataset.action === 'next' ? 1 : -1))); this.render(); }
      }, { signal: this.abort.signal }));
      root.querySelectorAll<HTMLButtonElement>('[data-detail-toggle]').forEach(button => button.addEventListener('click', () => {
        this.stop();
        const figure = button.closest<HTMLElement>('figure')!;
        const detail = figure.dataset.detail !== 'true';
        figure.dataset.detail = String(detail);
        button.setAttribute('aria-pressed', String(detail));
        button.textContent = detail ? 'Volver al encuadre completo' : 'Ampliar zona de contacto';
      }, { signal: this.abort.signal }));
      document.addEventListener('visibilitychange', () => { if (document.hidden) this.stop(); }, { signal: this.abort.signal });
      this.reduced.addEventListener('change', this.preference, { signal: this.abort.signal });
      this.preference();
    }
    private preference = () => {
      this.root.dataset.reduced = String(this.reduced.matches);
      if (this.reduced.matches) this.stop();
      this.root.querySelector<HTMLButtonElement>('[data-action="play"]')!.disabled = this.reduced.matches;
      this.root.querySelector<HTMLElement>('.motion-note')!.textContent = this.reduced.matches ? 'Movimiento reducido: consulta los pasos manualmente, sin transiciones.' : 'Sin reproducción automática. Puedes avanzar con los botones de cada paso.';
      this.render();
    };
    private schedule() {
      this.timer = setTimeout(() => {
        if (!this.root.isConnected || !this.playing) return;
        if (this.step < 3) { this.step++; this.render(); this.schedule(); }
        else this.stop();
      }, 9000);
    }
    private stop() { clearTimeout(this.timer); this.playing = false; this.render(); }
    private render() {
      this.root.dataset.playing = String(this.playing);
      this.root.querySelectorAll<HTMLElement>('[data-panel]').forEach((panel, index) => { panel.dataset.active = String(index === this.step); });
      this.root.querySelectorAll<HTMLButtonElement>('[data-step]').forEach(button => button.setAttribute('aria-pressed', String(Number(button.dataset.step) === this.step)));
      this.root.querySelector<HTMLButtonElement>('[data-action="prev"]')!.disabled = this.step === 0;
      this.root.querySelector<HTMLButtonElement>('[data-action="next"]')!.disabled = this.step === 3;
      this.root.querySelector<HTMLButtonElement>('[data-action="play"]')!.textContent = this.playing ? 'Pausar' : this.step === 3 ? 'Repetir secuencia' : 'Ver secuencia';
      const position = this.root.querySelector<HTMLElement>('.position')!;
      position.setAttribute('aria-live', this.playing ? 'off' : 'polite');
      position.textContent = `Paso ${this.step + 1} de 4`;
    }
    disconnect() { clearTimeout(this.timer); this.abort.abort(); }
  }
  const instances = new Map<HTMLElement, DentalGuide>();
  function initialize() { document.querySelectorAll<HTMLElement>('[data-dental-guide]').forEach(root => { if (!instances.has(root)) instances.set(root, new DentalGuide(root)); }); }
  document.addEventListener('astro:before-swap', () => { instances.forEach(instance => instance.disconnect()); instances.clear(); });
  document.addEventListener('astro:page-load', initialize);
  initialize();
</script>
