<script>
  class FeedingSequence {
    private step = 0;
    private playing = false;
    private timer: ReturnType<typeof setTimeout> | undefined;
    private aborter = new AbortController();
    private reduced = matchMedia('(prefers-reduced-motion: reduce)');
    constructor(private root: HTMLElement) {
      root.dataset.enhanced = 'true';
      root.querySelector<HTMLElement>('.asv-controls')!.hidden = false;
      root.querySelectorAll<HTMLButtonElement>('[data-action]').forEach(button => button.addEventListener('click', () => {
        if (button.dataset.action === 'play') {
          if (this.playing) this.stop();
          else if (!this.reduced.matches) { if (this.step === 2) this.step = 0; this.playing = true; this.render(); this.schedule(); }
        } else { this.stop(); this.step = Math.max(0, Math.min(2, this.step + (button.dataset.action === 'next' ? 1 : -1))); this.render(); }
      }, { signal: this.aborter.signal }));
      this.reduced.addEventListener('change', this.preference, { signal: this.aborter.signal });
      document.addEventListener('visibilitychange', this.visibility, { signal: this.aborter.signal });
      window.addEventListener('pagehide', this.stop, { signal: this.aborter.signal });
      this.preference();
    }
    private preference = () => {
      this.stop();
      this.root.querySelector<HTMLButtonElement>('[data-action="play"]')!.disabled = this.reduced.matches;
      this.root.querySelector<HTMLElement>('.asv-motion')!.textContent = this.reduced.matches ? 'Movimiento reducido: recorre los pasos con las flechas, sin animación.' : 'También puedes avanzar a tu ritmo con las flechas.';
    };
    private visibility = () => { if (document.hidden) this.stop(); };
    private stop = () => { clearTimeout(this.timer); this.playing = false; this.render(); };
    private schedule() {
      this.timer = setTimeout(() => {
        if (!this.root.isConnected || !this.playing) { this.stop(); return; }
        if (this.step < 2) { this.step++; this.render(); this.schedule(); }
        else this.stop();
      }, 6500);
    }
    private render() {
      this.root.dataset.step = String(this.step); this.root.dataset.playing = String(this.playing);
      this.root.querySelectorAll<HTMLElement>('[data-panel]').forEach((panel,i) => panel.dataset.active = String(i === this.step));
      const position = this.root.querySelector<HTMLElement>('.asv-position')!;
      position.setAttribute('aria-live', this.playing ? 'off' : 'polite'); position.textContent = `Paso ${this.step+1} de 3`;
      this.root.querySelector<HTMLButtonElement>('[data-action="prev"]')!.disabled = this.step === 0;
      this.root.querySelector<HTMLButtonElement>('[data-action="next"]')!.disabled = this.step === 2;
      this.root.querySelector<HTMLButtonElement>('[data-action="play"]')!.textContent = this.playing ? 'Pausar' : this.step === 2 ? 'Repetir secuencia' : 'Reproducir secuencia';
    }
    disconnect() { clearTimeout(this.timer); this.aborter.abort(); }
  }
  const instances = new Map<HTMLElement, FeedingSequence>();
  const recordAborters = new Map<HTMLElement, AbortController>();
  let printCopy: HTMLElement | null = null;
  function clearPrint() { printCopy?.remove(); printCopy = null; delete document.body.dataset.feedingPrint; }
  function initialize() {
    document.querySelectorAll<HTMLElement>('[data-feeding-sequence]').forEach(root => {
      if (!instances.has(root)) instances.set(root, new FeedingSequence(root));
    });
    document.querySelectorAll<HTMLElement>('[data-feeding-visual="registro"]').forEach(root => {
      if (recordAborters.has(root)) return;
      const aborter = new AbortController(); recordAborters.set(root, aborter);
      const button = root.querySelector<HTMLButtonElement>('.asv-print')!; button.hidden = false;
      button.addEventListener('click', () => {
        clearPrint();
        printCopy = root.cloneNode(true) as HTMLElement;
        printCopy.classList.add('asv-print-copy');
        printCopy.removeAttribute('aria-labelledby');
        printCopy.querySelectorAll('[id]').forEach(el => el.removeAttribute('id'));
        printCopy.querySelectorAll('label').forEach(el => el.removeAttribute('for'));
        printCopy.querySelectorAll('button,noscript').forEach(el => el.remove());
        const originals = root.querySelectorAll<HTMLInputElement | HTMLTextAreaElement>('input,textarea');
        printCopy.querySelectorAll('input,textarea').forEach((el,i) => {
          const value = document.createElement('p'); value.className = 'asv-printed-value'; value.textContent = originals[i].value || '________________________________'; el.replaceWith(value);
        });
        document.body.appendChild(printCopy); document.body.dataset.feedingPrint = 'true'; window.print();
      }, {signal:aborter.signal});
    });
  }
  window.addEventListener('afterprint', clearPrint);
  document.addEventListener('astro:before-swap', () => {
    instances.forEach(instance => instance.disconnect()); instances.clear();
    recordAborters.forEach(aborter => aborter.abort()); recordAborters.clear(); clearPrint();
  });
  document.addEventListener('astro:page-load', initialize);
  initialize();
</script>
